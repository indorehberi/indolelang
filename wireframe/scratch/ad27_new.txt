                  <tr>

          <!-- Add Branch Modal -->
          <div id="branch-modal" style="display:none; position:fixed; top:0; left:0; width:100%; height:100%; background:rgba(10,13,20,0.8); backdrop-filter:blur(8px); z-index:9999; justify-content:center; align-items:center; padding:1rem;">
            <div class="card" style="width:100%; max-width:500px; background:rgba(15, 23, 42, 0.95); border:1px solid var(--wf-border); border-radius:12px; padding:2rem; box-shadow: 0 20px 25px -5px rgba(0, 0, 0, 0.5); position:relative;">
              <button id="close-branch-modal" style="position:absolute; top:1rem; right:1rem; background:none; border:none; font-size:1.5rem; color:var(--wf-text-muted); cursor:pointer;">&times;</button>
              <div style="font-size:1.25rem; font-weight:bold; color:#fff; margin-bottom:1.5rem; border-bottom:1px solid var(--wf-border); padding-bottom:0.75rem;">Tambah Cabang Kantor Baru</div>
              
              <div class="form-group">
                <label class="form-label">Nama Cabang <span class="required">*</span></label>
                <input type="text" class="form-input" id="new-branch-name" placeholder="Contoh: Cabang Surabaya">
              </div>
              <div class="form-group">
                <label class="form-label">Kota <span class="required">*</span></label>
                <input type="text" class="form-input" id="new-branch-city" placeholder="Contoh: Surabaya">
              </div>
              <div class="form-group">
                <label class="form-label">Alamat Lengkap Gudang <span class="required">*</span></label>
                <input type="text" class="form-input" id="new-branch-address" placeholder="Contoh: Jl. Raya Darmo No. 88">
              </div>
              <div class="form-group">
                <label class="form-label">Nama Kepala Cabang <span class="required">*</span></label>
                <input type="text" class="form-input" id="new-branch-head" placeholder="Contoh: Bambang Utomo">
              </div>
              
              <div style="display:flex; justify-content:flex-end; gap:0.75rem; margin-top:2rem; border-top:1px solid var(--wf-border); padding-top:1.25rem;">
                <button id="btn-cancel-branch" class="btn btn-outline" style="margin-bottom:0;">Batal</button>
                <button id="btn-save-branch" class="btn btn-gold" style="margin-bottom:0;">Simpan Cabang</button>
              </div>
            </div>

          <script>
            document.addEventListener('DOMContentLoaded', function() {
              const defaultBranches = [
                { id: 'JKT-01', nama: 'Cabang Jakarta Pusat', kota: 'Jakarta', alamat: 'Jl. Gatot Subroto No. 45', kepala: 'M. Kurniawan', sesi: '3 Sesi', aset: '1.250 Aset', status: 'Aktif' },
                { id: 'BDG-02', nama: 'Cabang Bandung', kota: 'Bandung', alamat: 'Jl. Soekarno-Hatta No. 120', kepala: 'Dedi Suherman', sesi: '1 Sesi', aset: '480 Aset', status: 'Aktif' },
                { id: 'SUB-03', nama: 'Cabang Surabaya', kota: 'Surabaya', alamat: 'Jl. Raya Darmo No. 88', kepala: 'Bambang Utomo', sesi: '2 Sesi', aset: '890 Aset', status: 'Aktif' }
              ];
              
              function loadBranches() {
                const stored = localStorage.getItem('admin_branches');
                if (!stored) {
                  localStorage.setItem('admin_branches', JSON.stringify(defaultBranches));
                  return defaultBranches;
                }
                return JSON.parse(stored);
              }
              
              const tableBody = document.getElementById('branch-table-body');
              const searchInput = document.getElementById('search-branch');
              const modal = document.getElementById('branch-modal');
              const btnAdd = document.getElementById('btn-add-branch');
              const btnCancel = document.getElementById('btn-cancel-branch');
              const btnClose = document.getElementById('close-branch-modal');
              const btnSave = document.getElementById('btn-save-branch');
              
              function renderBranches(filter = '') {
                if (!tableBody) return;
                const branches = loadBranches();
                tableBody.innerHTML = '';
                
                const filtered = branches.filter(b => 
                  b.nama.toLowerCase().includes(filter.toLowerCase()) ||
                  b.kota.toLowerCase().includes(filter.toLowerCase()) ||
                  b.id.toLowerCase().includes(filter.toLowerCase()) ||
                  b.kepala.toLowerCase().includes(filter.toLowerCase())
                );
                
                if (filtered.length === 0) {
                  tableBody.innerHTML = '<tr><td colspan="9" class="text-center text-muted" style="padding:2rem;">Tidak ada cabang ditemukan.</td></tr>';
                  return;
                }
                
                filtered.forEach(b => {
                  const tr = document.createElement('tr');
                  tr.innerHTML = \`
                    <td class="fw-bold">\${b.id}</td>
                    <td class="fw-bold text-primary">\${b.nama}</td>
                    <td>\${b.kota}</td>
                    <td class="fs-sm text-muted">\${b.alamat}</td>
                    <td>\${b.kepala}</td>
                    <td>\${b.sesi}</td>
                    <td>\${b.aset}</td>
                    <td><span class="badge badge-success">\${b.status}</span></td>
                    <td>
                      <button class="btn btn-sm btn-outline text-danger btn-delete-branch" data-id="\${b.id}">Hapus</button>
                    </td>
                  \`;
                  tableBody.appendChild(tr);
                });
                
                // Bind delete event
                document.querySelectorAll('.btn-delete-branch').forEach(btn => {
                  btn.addEventListener('click', function() {
                    const id = this.getAttribute('data-id');
                    if (confirm('Apakah Anda yakin ingin menghapus cabang ' + id + '?')) {
                      let stored = loadBranches();
                      stored = stored.filter(b => b.id !== id);
                      localStorage.setItem('admin_branches', JSON.stringify(stored));
                      renderBranches(searchInput ? searchInput.value : '');
                    }
                  });
                });
              }
              
              if (searchInput) {
                searchInput.addEventListener('input', function() {
                  renderBranches(this.value);
                });
              }
              
              if (btnAdd) {
                btnAdd.addEventListener('click', function() {
                  modal.style.display = 'flex';
                });
              }
              
              function closeModal() {
                modal.style.display = 'none';
                document.getElementById('new-branch-name').value = '';
                document.getElementById('new-branch-city').value = '';
                document.getElementById('new-branch-address').value = '';
                document.getElementById('new-branch-head').value = '';
              }
              
              if (btnCancel) btnCancel.addEventListener('click', closeModal);
              if (btnClose) btnClose.addEventListener('click', closeModal);
              
              if (btnSave) {
                btnSave.addEventListener('click', function() {
                  const nama = document.getElementById('new-branch-name').value.trim();
                  const kota = document.getElementById('new-branch-city').value.trim();
                  const alamat = document.getElementById('new-branch-address').value.trim();
                  const kepala = document.getElementById('new-branch-head').value.trim();
                  
                  if (!nama || !kota || !alamat || !kepala) {
                    alert('Harap isi seluruh field bertanda bintang (*).');
                    return;
                  }
                  
                  const stored = loadBranches();
                  const nextNum = stored.length > 0 ? Math.max(...stored.map(b => parseInt(b.id.split('-')[1]) || 0)) + 1 : 1;
                  const nextId = 'CBG-0' + nextNum;
                  
                  const newBranch = {
                    id: nextId,
                    nama,
                    kota,
                    alamat,
                    kepala,
                    sesi: '0 Sesi',
                    aset: '0 Aset',
                    status: 'Aktif'
                  };
                  
                  stored.push(newBranch);
                  localStorage.setItem('admin_branches', JSON.stringify(stored));
                  
                  closeModal();
                  renderBranches(searchInput ? searchInput.value : '');
                  alert('Cabang ' + nama + ' berhasil ditambahkan!');
                });
              }
              
              // Initial render
              renderBranches();
            });
          </script>
